import { NextResponse } from 'next/server';

export const runtime = 'nodejs';
export const maxDuration = 300;
const MAX_BYTES = 25 * 1024 * 1024;
const LONG_SENTENCE_WORDS = 15;
const DANGLING_WORDS = new Set(['a', 'as', 'o', 'os', 'um', 'uma', 'uns', 'umas', 'de', 'do', 'da', 'dos', 'das', 'em', 'no', 'na', 'nos', 'nas', 'por', 'para', 'com', 'que', 'e', 'mas', 'não', 'porque', 'como', 'se', 'ao', 'à']);

function plainToken(value = '') {
  return value.normalize('NFD').replace(/\p{Diacritic}/gu, '').toLocaleLowerCase().replace(/[^\p{L}\p{N}]+/gu, '');
}

function isTerminal(value = '') {
  return /[.!?…]+[”"')\]}]*$/u.test(value.trim());
}

function isComma(value = '') {
  return /[,;:，、]+[”"')\]}]*$/u.test(value.trim());
}

function isDangling(value = '') {
  return DANGLING_WORDS.has(plainToken(value));
}

function alignWords(textTokens, timestampedWords) {
  const timed = timestampedWords
    .filter((word) => word?.word?.trim() && Number.isFinite(word.start) && Number.isFinite(word.end))
    .map((word) => ({ ...word, normalized: plainToken(word.word) }))
    .filter((word) => word.normalized);
  let cursor = 0;
  let previousEnd = 0;
  return textTokens.map((text) => {
    const target = plainToken(text);
    let found = -1;
    if (target) {
      const limit = Math.min(timed.length, cursor + 5);
      for (let i = cursor; i < limit; i++) {
        if (timed[i].normalized === target) { found = i; break; }
      }
    }
    if (found < 0) found = Math.min(cursor, Math.max(0, timed.length - 1));
    const match = timed[found];
    if (timed.length && target) {
      cursor = Math.min(timed.length, found + 1);
      previousEnd = match.end;
    }
    return { text, start: match?.start ?? previousEnd, end: match?.end ?? previousEnd };
  });
}

function capitalize(text) {
  const first = text.search(/[\p{L}\p{N}]/u);
  if (first < 0) return text;
  return text.slice(0, first) + text[first].toLocaleUpperCase('pt-BR') + text.slice(first + 1);
}

function finishPhrase(tokens, naturalBreak = false) {
  let text = tokens.map((token) => token.text).join(' ').replace(/\s+([,.;:!?…])/gu, '$1').trim();
  if (!text) return null;
  if (naturalBreak && !/[,;:]\s*[”"')\]}]*$/u.test(text)) text += ',';
  if (!/[.!?…,:;][”"')\]}]*$/u.test(text)) text += '.';
  text = capitalize(text);
  return {
    text,
    start: Math.max(0, tokens[0]?.start ?? 0),
    end: Math.max((tokens[0]?.start ?? 0) + 0.12, tokens.at(-1)?.end ?? 0),
  };
}

function sentenceTokens(textTokens) {
  const sentences = [];
  let current = [];
  for (const token of textTokens) {
    current.push(token);
    if (isTerminal(token)) {
      sentences.push({ tokens: current, complete: true });
      current = [];
    }
  }
  if (current.length) sentences.push({ tokens: current, complete: false });
  return sentences;
}

function mergeFragments(sentences) {
  for (let i = 0; i < sentences.length; i++) {
    const sentence = sentences[i];
    const count = sentence.tokens.length;
    const final = sentence.tokens.at(-1)?.text || '';
    const incomplete = !sentence.complete;
    const endsOnConnector = isDangling(final.replace(/[.!?…,:;]+[”"')\]}]*$/u, ''));
    if (count > 3 || (!incomplete && !endsOnConnector)) continue;
    if (i + 1 < sentences.length) {
      sentences[i + 1].tokens = [...sentence.tokens, ...sentences[i + 1].tokens];
      sentences[i + 1].complete = sentences[i + 1].complete;
      sentences.splice(i, 1);
      i--;
    } else if (i > 0) {
      sentences[i - 1].tokens.push(...sentence.tokens);
      sentences[i - 1].complete = sentence.complete;
      sentences.splice(i, 1);
      i--;
    }
  }
  return sentences;
}

function chooseLongBreak(tokens) {
  const total = tokens.length;
  if (total <= LONG_SENTENCE_WORDS) return null;
  const commaCandidates = [];
  const pauseCandidates = [];
  for (let i = 0; i < total - 3; i++) {
    const wordsBefore = i + 1;
    if (wordsBefore < 7 || wordsBefore > 17 || isDangling(tokens[i].text)) continue;
    if (isComma(tokens[i].text)) commaCandidates.push({ index: i, distance: Math.abs(15 - wordsBefore) });
    const gap = tokens[i + 1].start - tokens[i].end;
    if (gap >= 0.62) pauseCandidates.push({ index: i, distance: Math.abs(15 - wordsBefore), gap });
  }
  const choices = commaCandidates.length ? commaCandidates : pauseCandidates;
  if (!choices.length) return null;
  choices.sort((a, b) => a.distance - b.distance || (b.gap || 0) - (a.gap || 0));
  return { index: choices[0].index, natural: commaCandidates.length === 0 };
}

function splitLongSentence(sentence) {
  const result = [];
  let remaining = sentence.tokens;
  while (remaining.length > LONG_SENTENCE_WORDS) {
    const boundary = chooseLongBreak(remaining);
    if (!boundary) break; // Keep a complete sentence intact when there is no natural phrase boundary.
    result.push(finishPhrase(remaining.slice(0, boundary.index + 1), boundary.natural));
    remaining = remaining.slice(boundary.index + 1);
  }
  if (remaining.length) result.push(finishPhrase(remaining, false));
  return result.filter(Boolean);
}

function makeSentences(transcript, timestampedWords) {
  // Transcript punctuation defines sentences; word timestamps are used only for timing and long-sentence pauses.
  const textTokens = (String(transcript || '').match(/\S+/gu) || []);
  if (!textTokens.length) return [];
  const aligned = alignWords(textTokens, timestampedWords || []);
  const sentences = mergeFragments(sentenceTokens(aligned));
  return sentences.flatMap(splitLongSentence);
}

export async function POST(request) {
  const apiKey = process.env.OPENAI_API_KEY;
  if (!apiKey) {
    return NextResponse.json({ error: 'Transcription is not configured yet. Add OPENAI_API_KEY to the app’s .env.local file, then restart the server.' }, { status: 503 });
  }
  try {
    const data = await request.formData();
    const file = data.get('file');
    const language = data.get('language');
    if (!(file instanceof File) || !file.size) return NextResponse.json({ error: 'Choose a non-empty audio or video file first.' }, { status: 400 });
    if (file.size > MAX_BYTES) return NextResponse.json({ error: 'This file is over the 25 MB transcription limit. Compress or trim it, then try again.' }, { status: 413 });
    if (!/\.(mp3|mp4|mpeg|mpga|m4a|wav|webm|flac|ogg)$/i.test(file.name)) {
      return NextResponse.json({ error: 'Use an MP3, MP4, WAV, M4A, WEBM, FLAC, or OGG file.' }, { status: 415 });
    }

    const form = new FormData();
    form.append('file', new Blob([await file.arrayBuffer()], { type: file.type || 'application/octet-stream' }), file.name);
    form.append('model', 'whisper-1');
    form.append('response_format', 'verbose_json');
    form.append('timestamp_granularities[]', 'word');
    if (language && language !== 'auto') form.append('language', String(language).split('-')[0]);

    const response = await fetch('https://api.openai.com/v1/audio/transcriptions', {
      method: 'POST', headers: { Authorization: `Bearer ${apiKey}` }, body: form,
      signal: AbortSignal.timeout(240_000),
    });
    const result = await response.json().catch(() => ({}));
    if (!response.ok) {
      return NextResponse.json({ error: result?.error?.message || 'The speech-to-text service could not process this file.' }, { status: response.status === 401 ? 502 : response.status });
    }
    const words = (result.words || []).map(({ word, start, end }) => ({ word, start, end }));
    const segments = makeSentences(result.text || '', words);
    if (!segments.length && result.text) {
      return NextResponse.json({ error: 'A transcript was returned, but usable word timestamps were missing. Try another file or check the transcription service response.' }, { status: 502 });
    }
    return NextResponse.json({ text: result.text || '', duration: Number(result.duration) || segments.at(-1)?.end || 0, segments, words });
  } catch (error) {
    const message = error?.name === 'TimeoutError' ? 'Transcription took too long. Try a shorter audio clip.' : 'Could not transcribe this file. Check your connection and try again.';
    return NextResponse.json({ error: message }, { status: 500 });
  }
}
